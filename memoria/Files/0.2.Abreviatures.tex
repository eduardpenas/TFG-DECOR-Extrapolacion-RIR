\chapter*{Lista de abreviaturas}

\begin{description}
	\item[BIRD] \textit{Big Impulse Response Dataset} — corpus de respuestas impulsionales empleado en validación.
	\item[DDSP] \textit{Differentiable Digital Signal Processing} — Procesamiento Digital de Señal Diferenciable.
	\item[DECOR] \textit{Deep Exponential Completion Of Room Impulse Responses} — arquitectura propuesta para completación de RIR.
	\item[DRR] \textit{Direct-to-Reverberant Ratio} — Relación Directo-Reverberante.
	\item[EDC] \textit{Energy Decay Curve} — Curva de Decaimiento de Energía.
	\item[FINS] \textit{Filtered Noise Shaping} — arquitectura de referencia para síntesis de reverberación.
	\item[FIR] \textit{Finite Impulse Response} — filtro de respuesta al impulso finita.
	\item[GPU] \textit{Graphics Processing Unit} — unidad de procesamiento gráfico.
	\item[ISM] \textit{Image Source Method} — Método de Fuentes Imagen para simulación acústica.
	\item[ISO] \textit{International Organization for Standardization} — organismo de estandarización internacional.
	\item[LTI] \textit{Linear Time-Invariant} — sistema lineal e invariante en el tiempo.
	\item[MAE] \textit{Mean Absolute Error} — error absoluto medio.
	\item[MAPE] \textit{Mean Absolute Percentage Error} — error porcentual absoluto medio.
	\item[MLP] \textit{Multi-Layer Perceptron} — perceptrón multicapa.
	\item[MSE] \textit{Mean Squared Error} — error cuadrático medio.
	\item[MSTFT] \textit{Multi-Resolution Short-Time Fourier Transform} — pérdida espectral multirresolución.
	\item[PINN] \textit{Physics-Informed Neural Network} — red neuronal informada por la física.
	\item[RIR] \textit{Room Impulse Response} — Respuesta al Impulso de Sala.
	\item[RMSE] \textit{Root Mean Squared Error} — raíz del error cuadrático medio.
	\item[STFT] \textit{Short-Time Fourier Transform} — Transformada de Fourier de tiempo corto.
	\item[T\textsubscript{60}] \textit{Reverberation Time} — tiempo de reverberación para un decaimiento de 60~dB.
\end{description}
