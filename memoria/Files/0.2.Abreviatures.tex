\chapter*{Lista de abreviaturas}

\begin{description}
	\item[ESEIAAT] Escola Superior d'Enginyeries Industrial, Aeroespacial i Audiovisual de Terrassa \cite{eseiaat}.
	\item[RIR] \textit{Room Impulse Response} — Respuesta al Impulso de Sala.
	\item[EDC] \textit{Energy Decay Curve} — Curva de Decaimiento de Energía.
	\item[DECOR] \textit{DEep COmplete Reverb} — nombre del sistema propuesto.
	\item[BIRD] \textit{Big Impulse Response Dataset} — dataset de referencia de RIR reales.
	\item[ISM] \textit{Image Source Method} — Método de Fuentes Imagen.
	\item[LSTM] \textit{Long Short-Term Memory} — red neuronal recurrente con memoria a largo plazo.
	\item[MSTFT] \textit{Multi-resolution Short-Time Fourier Transform} — función de pérdida multiresolución.
	\item[MLP] \textit{Multi-Layer Perceptron} — perceptrón multicapa.
	\item[FIR] \textit{Finite Impulse Response} — filtro de respuesta al impulso finita.
	\item[PINN] \textit{Physics-Informed Neural Network} — red neuronal informada por la física.
	\item[AR] \textit{Augmented Reality} — Realidad Aumentada.
	\item[VR] \textit{Virtual Reality} — Realidad Virtual.
	\item[RT60] Tiempo de reverberación — tiempo (en segundos) para que el sonido decaiga 60~dB.
	\item[EDT] \textit{Early Decay Time} — tiempo de decaimiento temprano (primeros 10~dB).
	\item[T20] Tiempo de reverberación estimado a partir de los primeros 20~dB de decaimiento.
	\item[C50] \textit{Clarity} (50~ms) — medida de claridad acústica.
\end{description}
